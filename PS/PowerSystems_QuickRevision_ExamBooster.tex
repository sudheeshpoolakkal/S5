\subsection{2. Grid Energy Storage Systems (BESS, PHES, Flywheels, Supercapacitors)}
\begin{itemize}
  \item \textbf{Battery Energy Storage (BESS):} Lithium-Ion (high energy density, 85--95\% round-trip efficiency), Lead-Acid, Flow Batteries. Fast sub-second response for frequency regulation.
  \item \textbf{Pumped Hydro (PHES):} Largest grid storage ($>90\%$ global); round-trip efficiency $\eta \approx 70-80\%$; pumps water to upper reservoir during off-peak and generates during peak.
  \item \textbf{Flywheels \& Supercapacitors:} High power density, fast charge/discharge (milliseconds), millions of cycles; ideal for grid inertia and momentary voltage sag support.
\end{itemize}

\subsection{3. Load Dynamics \& Economic Dispatch (All Numerical Formulas)}

\begin{tcolorbox}[formulabox, title=Master Formulas for Load Curves \& Plant Economics]
\small
\begin{minipage}[t]{0.48\textwidth}
\begin{enumerate}
  \item $\text{Connected Load (CL)} = \sum \text{Rated capacities of all connected loads}$
  \item $\text{Maximum Demand (MD)} = \text{Peak instantaneous power demand}$
  \item $\text{Demand Factor} = \frac{\text{Maximum Demand}}{\text{Connected Load}} \le 1$
  \item $\text{Average Load} = \frac{\text{Total Energy Generated in Time } T}{\text{Time Duration } T} = \frac{\text{Area under Load Curve}}{24\,\text{hours}}$
  \item $\text{Load Factor (LF)} = \frac{\text{Average Load}}{\text{Maximum Demand}} = \frac{\text{Energy Generated in } T}{\text{MD} \times T} \le 1$
\end{enumerate}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\begin{enumerate}[resume]
  \item $\text{Diversity Factor} = \frac{\sum \text{Individual Maximum Demands}}{\text{Coincident System Maximum Demand}} \ge 1$
  \item $\text{Capacity Factor (CF)} = \frac{\text{Average Load}}{\text{Plant Rated Capacity}} = \text{LF} \times \frac{\text{MD}}{\text{Capacity}}$
  \item $\text{Plant Use Factor (UF)} = \frac{\text{Energy Generated in } T}{\text{Plant Capacity} \times \text{Operating Hours } t_{\text{op}}}$
  \item $\text{Reserve Capacity} = \text{Plant Rated Capacity} - \text{Maximum Demand}$
  \item $\text{Straight Line Depreciation:} \quad D = \frac{P - S}{n} \quad [\text{Rs/year}]$
  \item $\text{Sinking Fund:} \quad q = (P - S) \left[ \frac{r}{(1+r)^n - 1} \right] \quad [\text{Rs/year}]$
\end{enumerate}
\end{minipage}
\end{tcolorbox}

\begin{tcolorbox}[trapbox, title=Key Exam Traps \& Quick Facts for Module 1]
\footnotesize
$\bullet$ \textbf{Diversity Factor is ALWAYS $\ge 1$!} Higher diversity factor decreases coincident maximum demand, reducing required capital plant capacity and lowering generation cost per unit.\\
$\bullet$ \textbf{Base Load vs Peak Load Plants:} Base load (Coal, Nuclear, Run-of-river Hydro) have high capital cost, low running cost, and run at $100\%$ capacity. Peak load (Gas Turbine, Pumped Storage, Diesel) have low capital cost, high running cost, and quick startup time.
\end{tcolorbox}

\subsection{1. Transmission Line Inductance \& Capacitance (Formula Summary)}

\begin{tcolorbox}[formulabox, title=Inductance and Capacitance Exact Working Formulas]
\small
\textbf{1. Self GMR ($D_s$) and Internal Inductance:}
\begin{itemize}
  \item Solid Conductor Internal Inductance: $L_{\text{int}} = \frac{\mu_0}{8\pi} = \frac{1}{2} \times 10^{-7}\,\text{H/m}$ \quad (\textbf{Independent of conductor radius!})
  \item Fictitious Self GMR for Inductance: $D_s = r' = r \cdot e^{-1/4} = 0.7788 \cdot r$ \quad (Accounts for internal flux linkages).
  \item For Capacitance: **ALWAYS USE PHYSICAL RADIUS $r$** (Electrostatic charge resides exclusively on outer surface).
\end{itemize}

\textbf{2. 3-Phase Single Circuit Lines (Transposed):}
\begin{equation*}
  D_m = \sqrt[3]{D_{12} D_{23} D_{31}}, \qquad L = 2 \times 10^{-7} \ln\left(\frac{D_m}{D_s}\right)\,\text{H/m/phase}, \qquad C_n = \frac{2\pi\epsilon_0}{\ln(D_m / r)}\,\text{F/m/phase}
\end{equation*}

\textbf{3. 3-Phase Double Circuit Lines (Inverted Hexagonal $a-b-c$ vs $c'-b'-a'$):}
\begin{align*}
  D_{sa} &= \sqrt{r' D_{aa'}}, \quad D_{sb} = \sqrt{r' D_{bb'}}, \quad D_{sc} = \sqrt{r' D_{cc'}} \implies D_s = \sqrt[3]{D_{sa} D_{sb} D_{sc}} \\
  D_{ab} &= \sqrt[4]{D_{ab} D_{ab'} D_{a'b} D_{a'b'}}, \quad D_{bc} = \sqrt[4]{D_{bc} D_{bc'} D_{b'c} D_{b'c'}}, \quad D_{ca} = \sqrt[4]{D_{ca} D_{ca'} D_{c'a} D_{c'a'}} \implies D_m = \sqrt[3]{D_{ab} D_{bc} D_{ca}} \\
  L &= 2 \times 10^{-7} \ln\left(\frac{D_m}{D_s}\right)\,\text{H/m/phase}, \qquad C_n = \frac{2\pi\epsilon_0}{\ln(D_m / D_{sc})}\,\text{F/m/phase} \quad (\text{with } D_{sca} = \sqrt{r D_{aa'}})
\end{align*}
\end{tcolorbox}

\begin{itemize}
  \item \textbf{Fundamental Properties:} Symmetric $\iff A = D$; Reciprocal $\iff AD - BC = 1$.
  \item \textbf{Sending End Matrix:} $V_s = A V_r + B I_r, \quad I_s = C V_r + D I_r$.
  \item \textbf{Percentage Voltage Regulation:} $\%VR = \frac{|V_s / A| - |V_r|}{|V_r|} \times 100\%$ \quad (\textbf{Do NOT forget to divide $V_s$ by $A$ for medium/long lines!}).
  \item \textbf{Transmission Efficiency:} $\eta = \frac{P_r}{P_s} \times 100\% = \frac{3 |V_{r,\text{ph}}| |I_r| \cos\phi_r}{3 \operatorname{Re}(V_s I_s^*)} \times 100\%$.
\end{itemize}

\subsection{3. Operational Phenomena (Key Exam Definitions \& Derivations)}

\begin{center}
\scriptsize
\begin{tabularx}{\textwidth}{L{2.5cm} L{3.8cm} L{4.5cm} X}
  \rowcolor{AnthDark}
  \color{white}\textbf{Phenomenon} & \color{white}\textbf{Core Definition / Cause} & \color{white}\textbf{Mathematical Expression} & \color{white}\textbf{Consequences \& Mitigation} \\
  \rowcolor{AnthCardBG} \textbf{Skin Effect} & Concentration of AC current near conductor outer periphery due to internal flux eddy currents & Skin Depth $\delta = \sqrt{\frac{\rho}{\pi f \mu}}$ & Increases effective resistance ($R_{\text{ac}} > R_{\text{dc}}$). Mitigated by stranded ACSR conductors. \\
  \textbf{Ferranti Effect} & No-load / light-load receiving end voltage exceeds sending end voltage ($V_r > V_s$) due to capacitive charging current & $\Delta V = V_r - V_s \approx \frac{V_r \omega^2 l^2 L C}{2} > 0$ & Overvoltage stress on insulation; mitigated by connecting **Shunt Reactors** at receiving end. \\
  \rowcolor{AnthCardBG} \textbf{Corona} & Dielectric breakdown of air surrounding conductor when surface gradient exceeds $g_0 = 21.2\,\text{kV/cm (rms)}$ & $V_d = m_0 g_0 \delta r \ln(d/r)$ \newline $P = \frac{242.2}{\delta}(f+25)\sqrt{\frac{r}{d}}(V-V_d)^2 \times 10^{-5}$ & Pale-violet glow, hissing noise, $\text{O}_3$ corrosion, power loss. Mitigated by bundled conductors. \\
  \textbf{Surge Impedance Loading (SIL)} & Power delivered to purely resistive load equal to surge impedance $Z_c = \sqrt{L/C}$ & $\text{SIL} = \frac{V_{\text{L-L}}^2}{Z_c}$ \newline ($Z_c \approx 400\,\Omega$ overhead, $40\,\Omega$ cable) & At SIL, $Q_C = Q_L$, flat voltage profile ($V_s = V_r$). $P < \text{SIL} \to$ voltage rise; $P > \text{SIL} \to$ voltage drop. \\
\end{tabularx}
\end{center}

\subsection{4. Insulators \& String Efficiency (Quick Formula Guide)}
\begin{itemize}
  \item \textbf{Disc Voltages (3-Disc String with shunt ratio $k = C_1 / C$):}
    \begin{equation*}
      V_2 = V_1 (1 + k), \qquad V_3 = V_1 (1 + 3k + k^2), \qquad V = V_1 (3 + 4k + k^2)
    \end{equation*}
  \item \textbf{Maximum Voltage Stress:} Occurs across the disc \textbf{nearest to the conductor} ($V_3$).
  \item \textbf{String Efficiency:} $\eta = \frac{V}{n \times V_n} \times 100\%$.
  \item \textbf{Methods of Improvement:} Longer cross-arms (reduces $k$), Capacitance Grading ($C_1 < C_2 < C_3$), Guard Ring (injects current from line to achieve $100\%$ uniform voltage distribution).
\end{itemize}
