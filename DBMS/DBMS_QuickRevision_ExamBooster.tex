\section{MODULE 1: DBMS Architecture, ER Conceptual Modeling \& Physical Storage}
%% ─────────────────────────────────────────────────────────────────────────────

\subsection{1. Three-Schema Architecture \& Data Independence}
\begin{itemize}
  \item \textbf{Three Levels:} \textbf{External Level} (Tailored user views) $\to$ \textbf{Conceptual Level} (Community logical schema: entities, relationships, constraints) $\to$ \textbf{Internal Level} (Physical storage formats, block layout, access paths/indexes).
  \item \textbf{Logical Data Independence:} Ability to alter the \textbf{conceptual schema} (e.g., adding/deleting attributes or entities) without rewriting external schemas or application programs. \emph{(Harder to achieve)}.
  \item \textbf{Physical Data Independence:} Ability to alter the \textbf{internal schema} (e.g., building a B+ tree index, changing physical file organization) without changing the conceptual or external schemas. \emph{(Easy to achieve)}.
\end{itemize}

\subsection{3. Physical Storage \& Indexing Formulas (Guaranteed Numerical Question)}

\begin{tcolorbox}[formulabox, title=Master Formulas for File Storage \& Index Calculations]
\small
\begin{minipage}[t]{0.48\textwidth}
\begin{enumerate}
  \item $\text{Blocking Factor } bfr = \lfloor B / R \rfloor \quad [\text{records/block}]$
  \item $\text{Total Data Blocks } b = \lceil r / bfr \rceil \quad [\text{blocks}]$
  \item $\text{Wasted Block Space} = B - (bfr \times R) \quad [\text{bytes}]$
  \item $\text{Heap Linear Search Cost} = \lfloor b / 2 \rfloor \quad [\text{average I/Os}]$
  \item $\text{Primary Index (Sparse):}$ One entry per data block ($b$ entries); $R_i = V_K + P_B$; $bfr_i = \lfloor B / R_i \rfloor$; $b_i = \lceil b / bfr_i \rceil$; Binary Search Cost $= \lceil \log_2(b_i) \rceil + 1$.
\end{enumerate}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\begin{enumerate}[resume]
  \item $\text{Secondary Index (Dense on Key):}$ One entry per data record ($r$ entries); $b_i = \lceil r / bfr_i \rceil$; Search Cost $= \lceil \log_2(b_i) \rceil + 1$.
  \item $\text{Multi-Level Index Tree:}$ Fan-out $fo = bfr_i$;
  \begin{itemize}
    \item Level 1: $b_1 = \lceil b / fo \rceil$
    \item Level 2: $b_2 = \lceil b_1 / fo \rceil \dots$ until $b_x = 1$ (root)
  \end{itemize}
  \item $\text{Total Multi-Level Access Cost} = x + 1 \quad [\text{block accesses}]$.
\end{enumerate}
\end{minipage}
\end{tcolorbox}

\section{MODULE 2: Relational Model, 7-Step Synthesis, Relational Algebra \& Advanced SQL}
%% ─────────────────────────────────────────────────────────────────────────────

\subsection{1. Relational Keys \& Integrity Constraints (Quick Definitions)}
\begin{itemize}
  \item \textbf{Super Key ($SK$):} Uniquely identifies a row (may have extra attributes). \textbf{Candidate Key ($CK$):} Minimal super key.
  \item \textbf{Primary Key ($PK$):} Candidate key chosen for identification. \textbf{NEVER NULL!} \textbf{Alternate Key ($AK$):} $CK \setminus \{PK\}$.
  \item \textbf{Foreign Key ($FK$):} Field matching $PK$ of referenced table or NULL (Referential Integrity).
  \item \textbf{Entity Integrity:} $PK \neq \text{NULL}$. \textbf{Domain Integrity:} Values must belong to atomic domain.
\end{itemize}

\subsection{2. The 7-Step ER-to-Relational Synthesis Algorithm (Must Memorize!)}
\begin{enumerate}
  \item \textbf{Strong Entities:} Create table; flatten composite attributes into simple columns; $PK = \text{Entity } PK$.
  \item \textbf{Weak Entities:} Create table; import owner's $PK$ as $FK$; composite $PK = \{ \text{Owner } PK, \text{Partial Key} \}$.
  \item \textbf{Multivalued Attributes:} Create separate table with $\{ \text{Owner } PK_{\text{FK}}, \text{Attribute} \}$ as composite $PK$.
  \item \textbf{Binary 1:1 Relationships:} Foreign Key Approach $\to$ Place $PK$ of one side as $FK$ on the side with \textbf{total participation}.
  \item \textbf{Binary 1:N Relationships:} Place the $PK$ of the $1$-side as a \textbf{Foreign Key} inside the table on the \textbf{$N$-side}.
  \item \textbf{Binary M:N Relationships:} Create a separate \textbf{Junction Table} with composite $PK = \{ FK_1, FK_2 \}$.
  \item \textbf{Recursive 1:N / Ternary:} Add self-referencing $FK$ in same table / Create 3-way junction table with composite $PK$.
\end{enumerate}

\subsection{4. High-Yield SQL Templates for University Exams}

\begin{tcolorbox}[formulabox, title=Must-Know SQL Constructs]
\scriptsize
\begin{minipage}[t]{0.48\textwidth}
\textbf{1. Aggregation \& Grouping (WHERE vs HAVING):}
\begin{lstlisting}
SELECT DNo, COUNT(*) AS Total, AVG(Salary) AS AvgSal
FROM EMPLOYEE
WHERE Salary >= 20000.00
GROUP BY DNo
HAVING COUNT(*) >= 5 AND AVG(Salary) > 45000;
\end{lstlisting}
\textbf{2. Nested Subqueries (IN vs EXISTS):}
\begin{lstlisting}
-- Non-correlated
SELECT FName FROM EMPLOYEE 
WHERE DNo IN (SELECT DNumber FROM DEPARTMENT WHERE DName='EEE');

-- Correlated (Double NOT EXISTS for Division/ALL)
SELECT S.Name FROM STUDENT S WHERE NOT EXISTS (
    SELECT C.CourseID FROM COURSE C EXCEPT 
    SELECT E.CourseID FROM ENROLLS E WHERE E.RollNo = S.RollNo);
\end{lstlisting}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\textbf{3. View Creation:}
\begin{lstlisting}
CREATE VIEW HighEarners AS 
SELECT SSN, FName, Salary FROM EMPLOYEE WHERE Salary > 60000;
\end{lstlisting}

CREATE TRIGGER validate_salary

    IF NEW.Salary < 15000 THEN 
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Salary below minimum!';
    END IF; 
END;
